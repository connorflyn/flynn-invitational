        function initRoomListener(roomCode) {

            if (firestoreUnsubscribe) firestoreUnsubscribe();


            // Strict Firestore path format rule: /artifacts/{appId}/public/data/{collectionName}/{docId}

            const roomDocRef = doc(db, 'artifacts', appId, 'public', 'data', 'matches', roomCode);


            firestoreUnsubscribe = onSnapshot(roomDocRef, (docSnap) => {

                    // Initialize new room document if non-existent

            });

        function calculateTeamStats(team) {

            const totalStrokes = team.holes.reduce((sum, h) => sum + (h.score || 0), 0);

            const totalPar = DEFAULT_COURSE.reduce((sum, h) => sum + h.par, 0);

            const diff = totalStrokes - totalPar;

            

            let diffFormatted = "E";

            if (diff > 0) diffFormatted = `+${diff}`;

            else if (diff < 0) diffFormatted = `${diff}`;


            return { totalStrokes, totalPar, diff, diffFormatted };

        }

        window.updateHoleScore = function(delta) {

            const team = localState.teams[activeScorecardTeamIdx];

            if (!team.holes[activeHoleIdx]) {

                team.holes[activeHoleIdx] = { score: DEFAULT_COURSE[activeHoleIdx].par, playerShots: [0,0,0,0] };

            }

            team.holes[activeHoleIdx].score = Math.max(1, team.holes[activeHoleIdx].score + delta);

            pushStateToCloud();

            renderActiveHoleCard();

            renderHolePills();

        };


        window.updatePlayerShot = function(playerIdx, delta) {

            const team = localState.teams[activeScorecardTeamIdx];

            if (!team.holes[activeHoleIdx].playerShots) {

                team.holes[activeHoleIdx].playerShots = [0,0,0,0];

            }

            const curr = team.holes[activeHoleIdx].playerShots[playerIdx] || 0;

            team.holes[activeHoleIdx].playerShots[playerIdx] = Math.max(0, curr + delta);

            pushStateToCloud();

            renderActiveHoleCard();

        };

            });


            summary += `\nTrack live at: ${window.location.href}`;

            copyToClipboard(summary, "Group chat summary copied!");

        };


        function copyToClipboard(text, successMsg) {

            if (navigator.clipboard && window.isSecureContext) {

                navigator.clipboard.writeText(text).then(() => showToast(successMsg)).catch(() => fallbackCopy(text, successMsg));

            } else {

                fallbackCopy(text, successMsg);

            }

        }


        function fallbackCopy(text, successMsg) {

            const textArea = document.createElement("textarea");

            textArea.value = text;

            document.body.appendChild(textArea);

            textArea.select();

            try {

                document.execCommand('copy');

                showToast(successMsg);

            } catch (err) {

                showToast("Clipboard copy failed");

            }

            document.body.removeChild(textArea);

        }


        function showToast(msg) {

            const toast = document.getElementById('toast-message');

            document.getElementById('toast-text').textContent = msg;

            toast.classList.remove('opacity-0', 'pointer-events-none');

            setTimeout(() => {

                toast.classList.add('opacity-0', 'pointer-events-none');

            }, 2500);

        }
